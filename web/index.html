Python SDK

from imageai import ImageAI
ai = ImageAI()                       # 无 Key 也能跑（本地演示模式），配 Key 直连真实大模型
print(ai.chat("你好"))                # image1.0 / image Max 快速直答
print(ai.chat("分析量子计算", model="imageUltra"))   # 思考系模型，附带剖析过程
ai.generate_image("赛博朋克风格的猫")
print(ai.translate("床前明月光", target_lang="英文"))
